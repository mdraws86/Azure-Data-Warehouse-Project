IF NOT EXISTS (SELECT * FROM sys.external_file_formats WHERE name = 'SynapseDelimitedTextFormat') 
	CREATE EXTERNAL FILE FORMAT [SynapseDelimitedTextFormat] 
	WITH ( FORMAT_TYPE = DELIMITEDTEXT ,
	       FORMAT_OPTIONS (
			 FIELD_TERMINATOR = ',',
			 USE_TYPE_DEFAULT = FALSE
			))
GO

IF NOT EXISTS (SELECT * FROM sys.external_data_sources WHERE name = 'rivvystorage_rivvyaccount_dfs_core_windows_net') 
	CREATE EXTERNAL DATA SOURCE [rivvystorage_rivvyaccount_dfs_core_windows_net] 
	WITH (
		LOCATION = 'abfss://rivvystorage@rivvyaccount.dfs.core.windows.net' 
	)
GO

CREATE EXTERNAL TABLE dbo.staging_station (
	[station_id] nvarchar(400),
	[name] nvarchar(400),
	[latitude] float,
	[longitude] float
	)
	WITH (
	LOCATION = 'public.station.txt',
	DATA_SOURCE = [rivvystorage_rivvyaccount_dfs_core_windows_net],
	FILE_FORMAT = [SynapseDelimitedTextFormat]
	)
GO
